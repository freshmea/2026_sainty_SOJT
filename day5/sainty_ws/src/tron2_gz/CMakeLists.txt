cmake_minimum_required(VERSION 3.10)
project(tron2_gz)

find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(yaml-cpp REQUIRED)

add_library(tron2_gz_control SHARED src/wheel_foot_system.cpp)
target_compile_features(tron2_gz_control PRIVATE cxx_std_17)
target_link_libraries(tron2_gz_control ignition-gazebo6::core yaml-cpp)
ament_target_dependencies(tron2_gz_control rclcpp sensor_msgs std_msgs geometry_msgs tf2_ros ament_index_cpp)
find_package(ignition-gui6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
find_package(Qt5 REQUIRED COMPONENTS Quick Widgets)
find_package(OpenGL REQUIRED)
qt5_add_resources(gui_resources gui/Tron2FrameSync.qrc)
add_library(Tron2FrameSync SHARED gui/Tron2FrameSync.cpp ${gui_resources})
target_compile_features(Tron2FrameSync PRIVATE cxx_std_17)
target_link_libraries(Tron2FrameSync ignition-gui6::ignition-gui6 ignition-plugin1::ignition-plugin1-register
                      Qt5::Quick Qt5::Widgets OpenGL::GL)

install(TARGETS Tron2FrameSync tron2_gz_control LIBRARY DESTINATION lib)
install(DIRECTORY launch worlds urdf meshes DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE NOTICE THIRD_PARTY_NOTICES.md DESTINATION share/${PROJECT_NAME})
ament_package()
