cmake_minimum_required(VERSION 3.10)
project(tron2_control)
find_package(ament_cmake REQUIRED)
foreach(dep hardware_interface controller_interface pluginlib rclcpp rclcpp_lifecycle sensor_msgs std_msgs realtime_tools)
  find_package(${dep} REQUIRED)
endforeach()
add_library(tron2_control_plugins SHARED src/system.cpp src/controller.cpp)
target_compile_features(tron2_control_plugins PRIVATE cxx_std_17)
ament_target_dependencies(tron2_control_plugins hardware_interface controller_interface pluginlib rclcpp rclcpp_lifecycle sensor_msgs std_msgs realtime_tools)
pluginlib_export_plugin_description_file(hardware_interface hardware.xml)
pluginlib_export_plugin_description_file(controller_interface controllers.xml)
install(TARGETS tron2_control_plugins LIBRARY DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
